A sample from the logging loop
weather_station.py PYTHON 
import time
import board
import adafruit_bme280.advanced as bme280

# BME280 over I2C: temperature, pressure, humidity
i2c = board.I2C()
sensor = bme280.Adafruit_BME280_I2C(i2c, address=0x77)
sensor.sea_level_pressure = 1013.25

with open("weather_log.csv", "a") as log:
    while True:
        stamp = time.strftime("%Y-%m-%d %H:%M:%S")
        t = sensor.temperature          # deg C
        p = sensor.pressure             # hPa
        h = sensor.relative_humidity    # %
        log.write(f"{stamp},{t:.2f},{p:.2f},{h:.2f}\n")
        log.flush()
        print(f"{stamp}  {t:.1f} C  {p:.1f} hPa  {h:.1f} %")
        time.sleep(60)
